cmake_minimum_required(VERSION 3.15)
project(polar_spectrum VERSION 0.1.1 LANGUAGES CXX)
add_library(polar_spectrum SHARED cpp/src/kernel.cpp)
target_compile_features(polar_spectrum PRIVATE cxx_std_17)
target_include_directories(polar_spectrum PUBLIC cpp/include)
if(MINGW)
  target_link_options(polar_spectrum PRIVATE -static-libgcc -static-libstdc++)
endif()
install(TARGETS polar_spectrum DESTINATION lib)
install(FILES cpp/include/polar_spectrum.h DESTINATION include)
